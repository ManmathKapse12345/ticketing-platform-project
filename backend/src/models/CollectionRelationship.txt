routes/
  auth.route.js            → /register, /login  (public, self-service only)
  organization.route.js    → /organizations/:id/invite, /organizations/:id/members, etc.
  invite.route.js          → /invites/:token, /invites/:token/accept-new, /invites/:token/accept-existing

controllers/
  auth.controller.js        → register, login
  organization.controller.js→ sendInvite, listMembers, removeMember, updateMemberRole
  invite.controller.js      → getInviteDetails, acceptInviteNewUser, acceptInviteExistingUser

services/
  auth.service.js           → registerCustomer, registerOwner, loginUser
  organization.service.js   → createInvite, addMemberToOrg, removeMemberFromOrg
  invite.service.js         → generateInviteToken, verifyInviteToken, consumeInvite
